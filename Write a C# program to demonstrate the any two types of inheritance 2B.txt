using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace Pract2B
{
    class grandparent
    {
        public string gname;
        public void getname()
        {
            Console.WriteLine("Enter grandparent name");
            gname = Console.ReadLine();
        }
    }

    class father : grandparent
    {
        public string fname;
        public void getfname()
        {
            Console.WriteLine("Enter father name");
            fname = Console.ReadLine();
        }
    }

    class child : father
    {
        public string cname;
        public void getcname()
        {
            getname();
            getfname();
            Console.WriteLine("Enter child name");
            cname = Console.ReadLine();
        }

        public void show()
        {
            Console.WriteLine("I am " + cname + " child of " + fname + " grandchild of " + gname);
        }
    }

    class Program
    {
        static void Main(string[] args)
        {
            Console.WriteLine("Multilevel Inheritance");
            child c = new child();
            c.getcname();
            c.show();

            Console.ReadLine();
        }
    }
}
