using System;
using System.Collections.Generic;
using System.Linq;
using System.Text;
using System.Threading.Tasks;

namespace Pract1B
{
    class Program
    {
        public static void Main(string[] args)
        {
            int n, num = 1;
            Console.WriteLine("Enter number of row");
            n = Convert.ToInt32(Console.ReadLine());
            
            for (int i = 1; i <= n; i++)
            {
                for (int j = 1; j <= i; j++)
                {
                    Console.Write(num + " ");
                    num++;
                }
                Console.WriteLine();
            }
            
            Console.ReadLine();
        }
    }
}
